# XÓA ĐOẠN NÀY:
def logic_triplet_loss(z_cnf, z_s, z_us, margin=1.0):
    """
    Triplet loss kiểu chuẩn:
        L = mean( max(0, margin + d_s - d_us) )
    d_s  = ||z_cnf - z_s||^2
    d_us = ||z_cnf - z_us||^2
    """
    d_s = ((z_cnf - z_s) ** 2).sum(dim=1)
    d_us = ((z_cnf - z_us) ** 2).sum(dim=1)
    return F.relu(margin + d_s - d_us).mean()

# THAY BẰNG:
def logic_triplet_loss(z_cnf, z_s, z_us):
    """
    Eq. (14) từ paper SymGraphAU:
        L_h = Σ ‖d_s‖² − Σ 1_{d_s > d_us} · ‖d_us‖²

    Term 1: luôn kéo G_cnf gần G_s (attractive, không margin)
    Term 2: chỉ đẩy G_cnf xa G_us khi d_s > d_us (hard-mining indicator)
    """
    d_s = ((z_cnf - z_s) ** 2).sum(dim=1)    # (B,)
    d_us = ((z_cnf - z_us) ** 2).sum(dim=1)  # (B,)
    indicator = (d_s > d_us).float()
    return d_s.mean() - (indicator * d_us).mean()
